\section{Introduction}
\textbf{ForjaTikZ} is a visual companion app in the \textbf{ForjaTeX App Suite} designed to build \textbf{TikZ} diagrams and \textbf{PGFPlots} scientific charts without memorizing raw \LaTeX{} syntax.

\section{Mode 1: Charts (PGFPlots)}
\begin{enumerate}
  \item \textbf{Chart Type:} Pick \textbf{Functions $f(x)$}, \textbf{Points / CSV} (with optional error bars), \textbf{Bar Chart (\texttt{ybar})}, or \textbf{3D Surface (\texttt{\textbackslash addplot3})}.
  \item \textbf{Axis Configuration:} Set title, X/Y labels, axis bounds (\texttt{xmin}, \texttt{xmax}, \texttt{ymin}, \texttt{ymax}), scale (\texttt{linear}, \texttt{semilogx}, \texttt{semilogy}, \texttt{loglog}), and grid mode.
  \item \textbf{Math Formulas:} Write expressions in \texttt{x} (e.g., \texttt{0.2*x\^{}2}, \texttt{sin(x)}, \texttt{exp(-x\^{}2)}). Use one-click preset buttons for common formulas. With \textbf{Use Radians} enabled (\texttt{trig format plots=rad}), \texttt{sin(x)} evaluates directly in radians.
  \item \textbf{Data \& Bar Charts:} In \textbf{Points / CSV}, enter one point per line as \texttt{x, y} or \texttt{x, y, error}. In \textbf{Bar Chart}, enter \texttt{Category=Value} (e.g., \texttt{Jan=15}).
\end{enumerate}

\section{Mode 2: Diagrams (TikZ)}
\begin{itemize}
  \item \textbf{Shapes (\texttt{shapes.geometric}):} Add Circles, Boxes, Diamonds, and Ellipses and drag them on the snap-to-grid canvas.
  \item \textbf{Connections:} Select a node, click \textbf{Connect Selected Node}, then click the target node. Customize edge labels and curvature (\texttt{bend left} / \texttt{bend right}).
  \item \textbf{CTAN Templates:} Insert starter code for \texttt{tikz-cd} (commutative diagrams), \texttt{automata} (state machines), and \texttt{circuitikz} (circuits).
\end{itemize}

\section{Export and Validation}
\begin{itemize}
  \item \textbf{Copy Code:} Copies the generated \texttt{\textbackslash begin\{tikzpicture\}...\textbackslash end\{tikzpicture\}} block to paste into \textbf{ForjaTeX IDE}.
  \item \textbf{Export .tex:} Saves the snippet to a \texttt{.tex} file.
  \item \textbf{Test LaTeX:} Runs local \texttt{pdflatex} (if installed) to verify that the generated TikZ/PGF code compiles cleanly.
\end{itemize}

\section{Glossary}
\begin{description}
  \item[TikZ / PGF] Native \LaTeX{} vector graphics language for diagrams and technical illustrations.
  \item[PGFPlots] Package built on top of TikZ for publication-ready 2D and 3D plots.
  \item[\texttt{trig format plots=rad}] PGFPlots option that interprets trigonometric arguments in radians instead of degrees.
\end{description}
