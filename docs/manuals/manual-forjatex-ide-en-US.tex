\section{Introduction and Requirements}
\textbf{ForjaTeX IDE} is a free, open-source (\textbf{GNU GPL v3.0}), offline-first desktop \LaTeX{} IDE designed for academic papers, theses, books, and technical reports.

\section{Getting Started and Compilation}
\begin{enumerate}
  \item Click \textbf{Open Folder}, \textbf{New Project} (to start from ABNT, Article, or Beamer templates), or \textbf{Overleaf .zip} to import an exported Overleaf archive.
  \item In the left Explorer, right-click any \code{.tex} file and choose \textbf{Set as main file} if your root document is not \code{main.tex}.
  \item Pick an engine (\code{pdflatex}, \code{xelatex}, \code{lualatex}, or \code{latexmk}) and click \textbf{Compile}.
  \item \textbf{Per-Project Sequential Quick Build:} Right-click the \textbf{Compile} button to assign a custom multi-step sequence (e.g., \code{pdflatex -> bibtex -> pdflatex x2}) saved individually per project.
\end{enumerate}

\section{Editor, SyncTeX, and Live Previews}
\begin{itemize}
  \item \textbf{Smart Autocompletion:} Type \code{\textbackslash cite\{} for bibliography keys, \code{\textbackslash ref\{} for project labels, and \code{:} for custom User Tags.
  \item \textbf{Hover Equation Preview:} Hover over \code{\$...\$} or \code{\$\$...\$\$} expressions to see a live KaTeX formula preview.
  \item \textbf{Bidirectional SyncTeX:} Single-click in the PDF to highlight the source file in Explorer; double-click to jump to the exact line.
  \item \textbf{LaTeX-Aware Spell \& Grammar Checker:} Checks natural-language prose in PT-BR or English while ignoring commands, math, and comments, offering one-click Quick Fixes (\code{Ctrl+.}).
\end{itemize}

\section{Built-in Visual Wizards}
\begin{itemize}
  \item \textbf{Table Studio:} Visual editor supporting \code{booktabs} and \code{tabularray}, cell merging, and CSV/Excel paste.
  \item \textbf{Matrix \& Preamble Wizard:} Supports \code{amsmath} (\code{pmatrix}, \code{bmatrix}, \code{cases}) and \code{nicematrix} with live KaTeX preview.
  \item \textbf{System Font Wizard:} Detects installed TTF/OTF system fonts (Windows \& Linux \code{fc-list}) or copies local font files for \code{fontspec} (XeLaTeX/LuaLaTeX).
  \item \textbf{Image Wizard \& Offline Quick Reference:} Guided figure insertion and categorized/tagged snippet library.
\end{itemize}

\section{Settings, Git, and AI}
\begin{itemize}
  \item \textbf{Compiler \& Project:} Custom MiKTeX/TeX Live/Tectonic paths, isolated \code{build/} directory, and auto-save intervals.
  \item \textbf{Editor \& Automation:} Switch between Dark and Soft Ice Light themes, change language (\textbf{PT-BR} / \textbf{EN}), open this PDF manual anytime, and manage Quick Build profiles.
  \item \textbf{Built-in Git:} Configure \code{user.name}, \code{user.email}, Personal Access Token (GitHub/GitLab), generate a \LaTeX{} \code{.gitignore}, Clone, Commit, Pull, and Push.
  \item \textbf{AI Assistant (BYOK):} Available as a tab next to \textbf{PDF Preview} using your own API key (Gemini, OpenAI, Claude, or local Ollama).
\end{itemize}

\section{Glossary}
\begin{description}
  \item[SyncTeX] Bidirectional synchronization between \code{.tex} source lines and positions in the compiled PDF.
  \item[Sequential Quick Build] Chained execution of build steps (such as \code{pdflatex} and \code{bibtex}) to resolve citations and tables of contents.
  \item[LaTeX-Aware] A checker that understands \LaTeX{} markup so commands and math are not flagged as spelling errors.
  \item[BYOK (Bring Your Own Key)] Architecture where users supply their own AI provider key stored locally.
\end{description}
