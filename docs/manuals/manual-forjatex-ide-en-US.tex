\section{Introduction and Requirements}
\textbf{ForjaTeX IDE} is a free, open-source (GNU GPL v3.0), offline-first desktop \LaTeX{} IDE built with Tauri v2, Rust, React, and Monaco Editor.

\subsection{System Requirements}
\begin{itemize}
  \item Windows (64-bit) or source build for Linux/macOS.
  \item A local \TeX{} distribution installed on your machine: \textbf{MiKTeX}, \textbf{TeX Live}, or \textbf{Tectonic}.
\end{itemize}

\section{Getting Started and Compilation}
\begin{enumerate}
  \item Click \textbf{Open Folder}, \textbf{New Project} (to start from ABNT, Article, or Beamer templates), or \textbf{Overleaf .zip} to import an exported Overleaf archive.
  \item In the left Explorer, right-click any \texttt{.tex} file and choose \textbf{Set as main file} if your root document is not \texttt{main.tex}.
  \item Pick an engine (\texttt{pdflatex}, \texttt{xelatex}, \texttt{lualatex}, or \texttt{latexmk}) and click \textbf{Compile}.
  \item \textbf{Per-Project Sequential Quick Build:} Right-click the \textbf{Compile} button to assign a custom multi-step sequence (e.g., \texttt{pdflatex $\to$ bibtex $\to$ pdflatex $\times 2$}) saved per project.
\end{enumerate}

\section{Editor, SyncTeX, and Live Previews}
\begin{itemize}
  \item \textbf{Smart Autocompletion:} Type \texttt{\textbackslash cite\{} for bibliography keys, \texttt{\textbackslash ref\{} for project labels, and \texttt{:} for custom User Tags.
  \item \textbf{Hover Equation Preview:} Hover over \texttt{\$...\$\,} or \texttt{\$\$...\$\$\,} expressions to see a live KaTeX formula preview.
  \item \textbf{Bidirectional SyncTeX:} Single-click in the PDF to highlight the source file in Explorer; double-click to jump to the exact line.
  \item \textbf{LaTeX-Aware Spell \& Grammar Checker:} Checks natural-language prose in PT-BR or English while ignoring commands, math, and comments, offering one-click Quick Fixes (\texttt{Ctrl+.}).
\end{itemize}

\section{Built-in Visual Wizards}
\begin{itemize}
  \item \textbf{Table Studio:} Visual editor supporting \texttt{booktabs} and \texttt{tabularray}, cell merging, and CSV/Excel paste.
  \item \textbf{Matrix \& Preamble Wizard:} Supports \texttt{amsmath} (\texttt{pmatrix}, \texttt{bmatrix}, \texttt{cases}) and \texttt{nicematrix} with live KaTeX preview.
  \item \textbf{System Font Wizard:} Detects installed TTF/OTF system fonts or copies local font files for \texttt{fontspec} (XeLaTeX/LuaLaTeX).
  \item \textbf{Image Wizard \& Offline Quick Reference:} Guided figure insertion and categorized/tagged snippet library.
\end{itemize}

\section{Settings, Git, and AI}
\begin{itemize}
  \item \textbf{Compiler \& Project:} Custom MiKTeX/TeX Live/Tectonic paths, isolated \texttt{build/} directory, and auto-save intervals.
  \item \textbf{Editor \& Automation:} Switch between Dark and Soft Ice Light themes, change language (\textbf{PT-BR} / \textbf{EN}), and manage Quick Build profiles.
  \item \textbf{Built-in Git:} Configure \texttt{user.name}, \texttt{user.email}, Personal Access Token (GitHub/GitLab), generate a \LaTeX{} \texttt{.gitignore}, Clone, Commit, Pull, and Push.
  \item \textbf{AI Assistant (BYOK):} Available as a tab next to \textbf{PDF Preview} using your own API key (Gemini, OpenAI, Claude, or local Ollama).
\end{itemize}

\section{Glossary}
\begin{description}
  \item[SyncTeX] Bidirectional synchronization between \texttt{.tex} source lines and positions in the compiled PDF.
  \item[Sequential Quick Build] Chained execution of build steps (such as \texttt{pdflatex} and \texttt{bibtex}) to resolve citations and tables of contents.
  \item[LaTeX-Aware] A checker that understands \LaTeX{} markup so commands and math are not flagged as spelling errors.
  \item[BYOK (Bring Your Own Key)] Architecture where users supply their own AI provider key stored locally.
\end{description}
